  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  phone TEXT,
  password TEXT NOT NULL,
  role TEXT DEFAULT 'customer', -- 'customer', 'seller', 'admin'
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Ensure columns exist if users table was created previously

ALTER TABLE public.users ADD COLUMN IF NOT EXISTS phone TEXT;

-- Create Index for Phone field on users table (Fast Search & Lookup)
CREATE INDEX IF NOT EXISTS idx_users_phone ON public.users (phone);

-- 2. Create Stores Table (Toko Penjual)
CREATE TABLE IF NOT EXISTS public.stores (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT REFERENCES public.users(id) ON DELETE CASCADE,
  store_name TEXT NOT NULL,
  description TEXT,
  address TEXT,
  phone TEXT,
  status TEXT DEFAULT 'pending', -- 'pending', 'approved', 'rejected'
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Create Products Table (Catalog)
CREATE TABLE IF NOT EXISTS public.products (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  store_id BIGINT REFERENCES public.stores(id) ON DELETE SET NULL,
  store_name TEXT DEFAULT 'Toko Nelayan Bahari Pak Bambang',
  name TEXT NOT NULL,
  price NUMERIC NOT NULL,
  category TEXT NOT NULL,
  stock INT DEFAULT 0,
  image TEXT,
  unit TEXT DEFAULT 'kg',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Ensure store columns exist if products table was created previously
ALTER TABLE public.products ADD COLUMN IF NOT EXISTS store_id BIGINT REFERENCES public.stores(id) ON DELETE SET NULL;
ALTER TABLE public.products ADD COLUMN IF NOT EXISTS store_name TEXT DEFAULT 'Toko Nelayan Bahari Pak Bambang';

-- 4. Create Orders Table
CREATE TABLE IF NOT EXISTS public.orders (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_name TEXT NOT NULL,
  total_amount NUMERIC NOT NULL,
  status TEXT DEFAULT 'Menunggu Konfirmasi',
  items JSONB DEFAULT '[]'::jsonb,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. Create User Activity & Visitor Demographics Logs Table
CREATE TABLE IF NOT EXISTS public.user_logs (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT REFERENCES public.users(id) ON DELETE SET NULL,
  user_name TEXT DEFAULT 'Pengunjung Website',
  action TEXT NOT NULL, -- 'login', 'register', 'purchase', 'page_view'
  ip_address TEXT,
  city TEXT,
  region TEXT,
  country TEXT,
  device_type TEXT, -- 'Mobile', 'Desktop', 'Tablet'
  metadata JSONB DEFAULT '{}'::jsonb,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
